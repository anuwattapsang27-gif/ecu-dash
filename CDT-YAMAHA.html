// Web Bluetooth Connect Implementation (Modified for Universal OBD2 / ELM327) async function connectOBD2() { try { updateLog("กำลังค้นหาอุปกรณ์ Bluetooth ELM327..."); bluetoothDevice = await navigator.bluetooth.requestDevice({ acceptAllDevices: true, optionalServices: [...UART_SERVICE_UUIDS, 'generic_access'] }); bluetoothDevice.addEventListener('gattserverdisconnected', onDisconnected); updateLog("กำลังเชื่อมต่อ GATT Server..."); gattServer = await bluetoothDevice.gatt.connect(); // ดึง Service ทั้งหมดที่อุปกรณ์ปล่อยออกมาแบบอัตโนมัติ (แก้ปัญหาไม่พบ Service UART) let services = await gattServer.getPrimaryServices(); if (!services || services.length === 0) { throw new Error("ไม่พบ Service บนตัว Bluetooth นี้"); } // เลือก Service ตัวแรกที่อุปกรณ์รองรับ let service = services[0]; updateLog("พบ Service แล้ว กำลังค้นหา Characteristic..."); // ดึง Characteristic ทั้งหมดภายใน Service นั้น เพื่อหาช่องทางรับ-ส่งข้อมูล (RX/TX) let characteristics = await service.getCharacteristics(); for (let characteristic of characteristics) { let properties = characteristic.properties; // ถ้าซัพพอร์ต Notify หรือ Indicate ให้ใช้เป็นขาอ่านข้อมูล (RX) if (properties.notify || properties.indicate) { rxChar = characteristic; } // ถ้าซัพพอร์ต Write หรือ WriteWithoutResponse ให้ใช้เป็นขาเขียนคำสั่ง (TX) if (properties.write || properties.writeWithoutResponse) { txChar = characteristic; } } // กรณีหาแยกไม่ได้ ให้ใช้ตัวเดียวกัน (Fallback) if (!rxChar && characteristics.length > 0) rxChar = characteristics[0]; if (!txChar && characteristics.length > 0) txChar = characteristics[0]; if (!rxChar || !txChar) { throw new Error("ไม่พบ Characteristic สำหรับรับ-ส่งข้อมูล"); } await rxChar.startNotifications(); rxChar.addEventListener('characteristicvaluechanged', handleBLEData); isConnected = true; document.getElementById('btn-connect').innerText = "🔌 ตัดการเชื่อมต่อ"; document.getElementById('btn-connect').style.background = "#ffb703"; updateLog("เชื่อมต่อแล้ว! กำลังตั้งค่า Protocol ELM327..."); await setupOBDProtocol(); } catch (error) { updateLog("ข้อผิดพลาด: " + error.message); alert("เชื่อมต่อไม่สำเร็จ: " + error.message + "\n\n*(หากใช้ iOS ต้องเปิดเว็บผ่านแอป Bluefy เท่านั้น)*"); disconnectOBD(); } }
